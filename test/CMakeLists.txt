macro(pythonize_bool truth_var var)
  if(${truth_var})
    set(${var} True)
  else()
    set(${var} False)
  endif()
endmacro()

function(configure_metavirt_lit_site_cfg input output)
  set(LIT_SITE_CFG_IN_HEADER
      "## Autogenerated for metavirt from ${input}\n## Do not edit!"
  )
  
  set(METAVIRT_PROJECT_DIR ${PROJECT_SOURCE_DIR})
  set(METAVIRT_LIBRARY_DIR ${CMAKE_BINARY_DIR}/lib)
  set(METAVIRT_PASS_DIR ${CMAKE_BINARY_DIR}/verifier/pass)
  
  set(METAVIRT_SCRIPT_DIR ${CMAKE_BINARY_DIR}/verifier/scripts)

  metavirt_target_generate_file(${input} ${output})
endfunction()

function(metavirt_add_lit_testsuite target comment)
  cmake_parse_arguments(ARG "" "" "PARAMS;DEPENDS;ARGS" ${ARGN})

  foreach(param ${ARG_PARAMS})
    list(APPEND TEST_PARAMS --param ${param})
  endforeach()

  add_custom_target(${target}
    COMMAND ${METAVIRT_LIT_EXEC} ${ARG_ARGS} ${TEST_PARAMS} ${ARG_UNPARSED_ARGUMENTS}
    COMMENT "${comment}"
    USES_TERMINAL
  )

  if (ARG_DEPENDS)
    add_dependencies(${target} ${ARG_DEPENDS})
  endif()
endfunction()

function(metavirt_add_lit_target)
  cmake_parse_arguments(ARG "" "" "SUITES" ${ARGN})

  foreach(suite IN LISTS ARG_SUITES)
    if("${suite}" STREQUAL "all")
      set(SUITE_PATH ${CMAKE_CURRENT_SOURCE_DIR})
      set(TARGET_NAME check-metavirt)
    else()
      set(SUITE_PATH ${CMAKE_CURRENT_SOURCE_DIR}/${suite})
      set(TARGET_NAME check-metavirt-${suite})
    endif()

    if(NOT EXISTS ${SUITE_PATH})
      message(WARNING "Could not find suitable lit test target at ${SUITE_PATH}")
      continue()
    endif()

    metavirt_add_lit_testsuite(${TARGET_NAME}
      "Running the lit suite metavirt::${suite}"
      ${SUITE_PATH}
      ARGS -v -j 1
      PARAMS metavirt_site_config=${CMAKE_CURRENT_BINARY_DIR}/lit.site.cfg
      DEPENDS ${METAVIRT_TEST_DEPENDS}
    )
  endforeach()
endfunction()
